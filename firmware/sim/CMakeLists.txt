# macOS SDL simulator: the same ui/ + model/ the device runs, in a 410x502
# window, fed by fixture JSON (including captured live /v1/nearby bodies).
cmake_minimum_required(VERSION 3.16)
project(gtfs_compass_sim C)

set(CMAKE_C_STANDARD 11)
add_compile_options(-Wall -Wextra -g)

set(FW ${CMAKE_CURRENT_SOURCE_DIR}/..)

find_package(SDL2 REQUIRED)

# LVGL from the pinned submodule; sim/lv_conf.h is a one-line wrapper around
# the repo's canonical firmware/lv_conf.h (LVGL's desktop build requires the
# file to sit at the project root).
add_compile_definitions(SIM_BUILD)
add_subdirectory(${FW}/third_party/lvgl ${CMAKE_BINARY_DIR}/lvgl)
# LVGL's SDL driver needs the SDL headers/libs too (Homebrew prefix).
get_filename_component(SDL2_PARENT_DIR "${SDL2_INCLUDE_DIRS}" DIRECTORY)
target_include_directories(lvgl PUBLIC ${SDL2_INCLUDE_DIRS} ${SDL2_PARENT_DIR})
target_link_libraries(lvgl PUBLIC ${SDL2_LIBRARIES})

add_executable(sim
  main.c
  ${FW}/components/model/model.c
  ${FW}/components/ui/ui_board.c
  ${FW}/components/ui/ui_views.c
  ${FW}/components/ui/ui_state.c
  ${FW}/components/ui/ui_nav.c
  ${FW}/components/ui/ui_input.c
  # IBM Plex Sans faces — generated by ../tools/genfonts.sh, committed
  ${FW}/components/ui/fonts/gc_plex_14.c
  ${FW}/components/ui/fonts/gc_plex_15.c
  ${FW}/components/ui/fonts/gc_plex_16.c
  ${FW}/components/ui/fonts/gc_plex_17.c
  ${FW}/components/ui/fonts/gc_plex_18.c
  ${FW}/components/ui/fonts/gc_plex_20.c
  ${FW}/components/ui/fonts/gc_plex_24.c
  ${FW}/components/ui/fonts/gc_plex_25.c
  ${FW}/components/ui/fonts/gc_plex_27.c
  ${FW}/components/ui/fonts/gc_plex_30.c
  ${FW}/components/ui/fonts/gc_plex_36.c
  ${FW}/components/ui/fonts/gc_plex_40.c
  ${FW}/components/ui/fonts/gc_plex_84.c
  ${FW}/third_party/cjson/cJSON.c
)
target_include_directories(sim PRIVATE
  ${FW}/components/model/include
  ${FW}/components/ui/include
  ${FW}/components/ui
  ${FW}/third_party/cjson
  ${SDL2_INCLUDE_DIRS}
)
target_link_libraries(sim lvgl ${SDL2_LIBRARIES})
target_compile_definitions(sim PRIVATE FIXTURE_DIR="${FW}/fixtures"
  CJSON_NESTING_LIMIT=48)

# U2 gesture-tracker tests: headless (no SDL window) — a scripted pointer
# indev driven through the real LVGL machinery. Run: ./build/test_input
add_executable(test_input
  test_input.c
  ${FW}/components/ui/ui_input.c
)
target_include_directories(test_input PRIVATE
  ${FW}/components/ui/include
  ${SDL2_INCLUDE_DIRS}
)
target_link_libraries(test_input lvgl ${SDL2_LIBRARIES})
